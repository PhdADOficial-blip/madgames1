
Principais tipos de cookies

• Cookies primários: Criados pelo próprio site que você está visitando para ajudar na navegação e funcionalidade básica.
• Cookies de terceiros: Criados por empresas parceiras (como redes sociais ou plataformas de anúncios) para rastrear o seu comportamento em vários sites diferentes e direcionar publicidade.
• Cookies de sessão: Temporários; são apagados automaticamente assim que você fecha o navegador.
• Cookies persistentes: Ficam salvos no seu aparelho até uma data de validade definida ou até você excluí-los manualmente.



Ideia para o projeto: O "Cupom de Primeira Visita"

Como o objetivo do site é apresentar a loja e gerar contatos, vamos usar o cookie para verificar se é a primeira vez que o cliente entra no site. Se for, o site mostra uma mensagem com um Cupom de Desconto (ex: GAMER10) para incentivar o cliente a rolar a página até o formulário de contato e mandar uma mensagem.
Se ele atualizar a página, o cookie "lembra" dele e não mostra o aviso repetido, evitando chatear o cliente.



// 1. Função para ler se o cookie já existe no navegador
function obterCookie(nome) {
    let nomeProcurado = nome + "=";
    let cookies = document.cookie.split(';');
    for(let i = 0; i < cookies.length; i++) {
        let c = cookies[i].trim();
        if (c.indexOf(nomeProcurado) == 0) return c.substring(nomeProcurado.length, c.length);
    }
    return "";
}

// 2. Lógica principal rodada ao carregar a página da loja de games
window.onload = function() {
    let visitou = obterCookie("usuario_ja_visitou");

    if (visitou === "") {
        // Se o cookie NÃO existe, é a primeira visita!
        // Aqui você mostra o alerta do cupom
        alert("🎮 Bem-vindo à nossa Loja Game! Use o cupom GAMER10 ao entrar em contato e ganhe 10% de desconto no seu primeiro jogo!");
        
        // Criamos o cookie para durar 7 dias, assim ele não vê o alert de novo se atualizar a página
        let data = new Date();
        data.setTime(data.getTime() + (7 * 24 * 60 * 60 * 1000));
        document.cookie = "usuario_ja_visitou=sim; expires=" + data.toUTCString() + "; path=/; SameSite=Lax; Secure";
    } else {
        // Se o cookie JÁ existe, o navegador apenas exibe uma mensagem discreta no console
        console.log("Usuário recorrente. O cookie evitou o alerta repetido.");
    }
};

// Função para fechar o modal na tela quando clicar no botão ou no X
function fecharModalGame() {
    document.getElementById("meuModalCookie").style.display = "none";
}

window.onload = function() {
    let visitou = obterCookie("usuario_ja_visitou");

    if (visitou === "") {
        // Se NÃO tem o cookie, exibe o modal estilizado mudando o display para flex
